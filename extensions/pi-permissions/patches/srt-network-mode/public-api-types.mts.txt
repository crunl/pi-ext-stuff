import type {
  NetworkConfig,
  NetworkMode,
  NetworkModeCapabilities,
  SandboxRuntimeConfig,
  SandboxWrapConfig,
} from "./package/dist/index.js";
import {
  NetworkConfigSchema,
  NetworkModeError,
  SandboxManager,
  SandboxRuntimeConfigSchema,
} from "./package/dist/index.js";

const mode: NetworkMode = "restricted";
const network: NetworkConfig = { mode, allowedDomains: [], deniedDomains: ["*"] };
const config: SandboxRuntimeConfig = {
  network,
  filesystem: { denyRead: [], allowWrite: [], denyWrite: [] },
};
const parsedMode: NetworkMode | undefined = SandboxRuntimeConfigSchema.parse(config).network.mode;
const parsedNetworkMode: NetworkMode | undefined = NetworkConfigSchema.parse(network).mode;
const wrap: SandboxWrapConfig = { network: { mode } };
const capabilities: NetworkModeCapabilities = SandboxManager.getNetworkModeCapabilities();
const version: 1 = capabilities.apiVersion;
void SandboxManager.wrapWithSandboxArgv("true", "/bin/bash", wrap);
void SandboxManager.wrapWithSandbox("true", "/bin/bash", wrap);
// @ts-expect-error Reject unknown mode at the public wrap seam.
void SandboxManager.wrapWithSandboxArgv("true", "/bin/bash", { network: { mode: "fake" } });
// @ts-expect-error Capabilities are read-only.
capabilities.modes.push("direct");
const error = new NetworkModeError("NETWORK_MODE_NOT_READY", "not initialized");
void [parsedMode, parsedNetworkMode, version, error];
